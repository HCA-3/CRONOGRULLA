643:     # ── MOTOR DE TRANSFORMACIÓN Y PROYECCIÓN 3D ────────────────────────────────
644:     def rotate_point_3d(self, x, y, z):
645:         """Rota un punto (x, y, z) en espacio euclidiano según Yaw y Pitch."""
646:         # 1. Rotación alrededor del eje Y (Yaw)
647:         cos_y = math.cos(self.yaw)
648:         sin_y = math.sin(self.yaw)
649:         x1 = x * cos_y - z * sin_y
650:         z1 = x * sin_y + z * cos_y
651:         
652:         # 2. Rotación alrededor del eje X (Pitch)
653:         cos_p = math.cos(self.pitch)
654:         sin_p = math.sin(self.pitch)
655:         y2 = y * cos_p - z1 * sin_p
656:         z2 = y * sin_p + z1 * cos_p
657:         
658:         return x1, y2, z2
